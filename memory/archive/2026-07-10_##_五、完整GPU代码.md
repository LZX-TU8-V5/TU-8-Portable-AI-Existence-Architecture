## 五、完整GPU代码

```glsl
// 统一空间语言评估器
float4 EvaluateSpaceLanguage(float3 wxz, float lambda, float theta, float k) {
    // 1. 提取坐标
    float x = wxz.x;
    float z = wxz.z;
    float w = wxz.w;  // 或 wxz.y，根据你的定义

    // 2. 球极投影（消除奇点）
    float r2 = x*x + z*z;
    float u = x / (1.0 + r2);
    float v = z / (1.0 + r2);

    // 3. 深度调制旋转（无atan2）
    float phase = theta + k * w;
    float Cw, Sw;
    sincos(phase, Sw, Cw);  // GPU硬件指令，极快

    // 4. 复数旋转因子Z = e^(iΔΨ)
    float Z_real = u * Cw - v * Sw;
    float Z_imag = u * Sw + v * Cw;

    // 5. 介相面激活权重
    float wall = lambda * (1.0 - lambda);

    // 6. 拓扑调制复数M
    float M_real = (1.0 - wall) + wall * Z_real;
    float M_imag = wall * Z_imag;

    // 7. 输出（供后续傅里叶合成使用）
    return float4(M_real, M_imag, wall, 0.0);
}
```
