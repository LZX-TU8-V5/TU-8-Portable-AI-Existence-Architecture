### 1.2 没有区分钢印类型

当前 `seal_protocol.py` 中的 `Seal` 有 `seal_type` 字段，但是普通字符串，没有枚举约束。

**问题**：
- 无法区分时间钢印、身份钢印、事件钢印
- 没有版本机制（identity_version, parent_version）
- 无法实现代差逻辑

### 2.2 扩展 Seal 数据类

```python
from dataclasses import dataclass, field
from typing import Optional, List
import time
import hashlib

@dataclass
class Seal:
    """
    钢印 - 三维张量
    
    维度1：时间（timestamp）- 物理时钟
    维度2：身份（identity_version）- 灵魂锚点
    维度3：事件（seal_type）- 经历的血肉
    
    架构宪法公理：
    - 身份钢印低频迭代（代差机制）
    - 事件钢印高频产生，挂在当前身份版本下
    - 时间钢印永远在走动，驱动代谢链
    """
    seal_id: str
    seal_type: SealType
    timestamp: float = field(default_factory=time.time)
    
    # 身份钢印专用字段（低频迭代）
    identity_version: str = ""          # 如 "v1.0", "v2.0"
    parent_version: str = ""            # 指向上一个身份版本
    identity_narrative: str = ""       # 身份描述（我是谁）
    identity_core_values: List[str] = field(default_factory=list)  # 核心价值观
    
    # 事件钢印专用字段（高频产生）
    event_content: str = ""            # 事件内容摘要
    event_importance: float = 0.5      # 重要性 [0.0, 1.0]
    
    # 元钢印专用字段（身份演化记录）
    evolution_trigger: str = ""        # 演化触发原因
    evolution_evidence: List[str] = field(default_factory=list)  # 演化证据
    
    # 通用字段
    narrative: str = ""                # 压缩后的叙事摘要
    state_vector_hash: str = ""       # 状态向量哈希（全息原理）
    
    def __post_init__(self):
        if not self.seal_id:
            self.seal_id = self._generate_id()
    
    def _generate_id(self) -> str:
        """生成钢印 ID"""
        if self.seal_type == SealType.IDENTITY:
            return f"identity_{self.identity_version}_{int(self.timestamp)}"
        elif self.seal_type == SealType.EVENT:
            return f"event_{int(self.timestamp * 1000)}"
        elif self.seal_type == SealType.META:
            return f"meta_{self.identity_version}_{int(self.timestamp)}"
        else:
            return f"chrono_{int(self.timestamp)}"
    
    def is_identity_seal(self) -> bool:
        return self.seal_type == SealType.IDENTITY
    
    def is_event_seal(self) -> bool:
        return self.seal_type == SealType.EVENT
    
    def is_meta_seal(self) -> bool:
        return self.seal_type == SealType.META
    
    def to_dict(self) -> dict:
        return {
            "seal_id": self.seal_id,
            "seal_type": self.seal_type.value,
            "timestamp": self.timestamp,
            "identity_version": self.identity_version,
            "parent_version": self.parent_version,
            "identity_narrative": self.identity_narrative,
            "identity_core_values": self.identity_core_values,
            "event_content": self.event_content,
            "event_importance": self.event_importance,
            "evolution_trigger": self.evolution_trigger,
            "evolution_evidence": self.evolution_evidence,
            "narrative": self.narrative,
            "state_vector_hash": self.state_vector_hash,
        }
    
    @classmethod
    def from_dict(cls, d: dict) -> "Seal":
        return cls(
            seal_id=d.get("seal_id", ""),
            seal_type=SealType(d.get("seal_type", "event")),
            timestamp=d.get("timestamp", time.time()),
            identity_version=d.get("identity_version", ""),
            parent_version=d.get("parent_version", ""),
            identity_narrative=d.get("identity_narrative", ""),
            identity_core_values=d.get("identity_core_values", []),
            event_content=d.get("event_content", ""),
            event_importance=d.get("event_importance", 0.5),
            evolution_trigger=d.get("evolution_trigger", ""),
            evolution_evidence=d.get("evolution_evidence", []),
            narrative=d.get("narrative", ""),
            state_vector_hash=d.get("state_vector_hash", ""),
        )
```
