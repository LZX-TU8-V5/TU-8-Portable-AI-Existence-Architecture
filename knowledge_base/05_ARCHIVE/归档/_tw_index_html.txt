 <script>
 const outputDisplay = document.getElementById('outputDisplay');
 const userInput = document.getElementById('userInput');
 const sendBtn = document.getElementById('sendBtn');
 const mockCheck = document.getElementById('mockCheck');

 function updateLayer(layerId, statusMsg, isActive = false) {
 const el = document.getElementById(`layer-${layerId}`);
 if (el) {
 const statusSpan = el.querySelector('.layer-status');
 statusSpan.innerText = statusMsg;
 if (isActive) {
 el.classList.add('active');
 }
 }
 }

 function appendText(text) {
 outputDisplay.innerText += text;
 outputDisplay.scrollTop = outputDisplay.scrollHeight;
 }

 async function sendMessage() {
 const text = userInput.value.trim();
 if (!text) return;

 userInput.value = '';
 userInput.disabled = true;
 sendBtn.disabled = true;
 sendBtn.innerText = '思考中...';
 outputDisplay.innerText = '> 正在连接 TU-8 核心...\n\n';
 
 document.querySelectorAll('.layer-item').forEach(el => {
 el.classList.remove('active', 'final');
 el.querySelector('.layer-status').innerText = '待机';
 });

 const useMock = mockCheck.checked;

 try {
 const response = await fetch('/api/chat', {
 method: 'POST',
 headers: { 'Content-Type': 'application/json' },
 body: JSON.stringify({ message: text, mock: useMock })
 });

 const reader = response.body.getReader();
 const decoder = new TextDecoder();

 while (true) {
 const { done, value } = await reader.read();
 if (done) break;

 const chunk = decoder.decode(value);
 const lines = chunk.split('\n');

 for (const line of lines) {
 if (line.startsWith('data: ')) {
 const jsonStr = line.slice(6);
 if (!jsonStr.trim()) continue;
 
 try {
 const data = JSON.parse(jsonStr);
 
 if (data.type === 'layer_update') {
 const layerData = data.data;
 updateLayer(layerData.layer, layerData.msg, true);
 if (layerData.layer === 'FINAL') {
 document.getElementById(`layer-${layerData.layer}`).classList.add('final');
 }
 } else if (data.type === 'text') {
 appendText(data.data);
 } else if (data.type === 'error') {
 appendText(`\n[系统错误]: ${data.data}`);
 }
 } catch (e) {
 console.error('解析错误', e);
 }
 }
 }
 }

 } catch (err) {
 outputDisplay.innerText += `\n[连接失败]: ${err.message}`;
 } finally {
 userInput.disabled = false;
 sendBtn.disabled = false;
 sendBtn.innerText = '发送指令';
 userInput.focus();
 }
 }

 sendBtn.addEventListener('click', sendMessage);
 userInput.addEventListener('keypress', (e) => {
 if (e.key === 'Enter') sendMessage();
 });
 </script>
